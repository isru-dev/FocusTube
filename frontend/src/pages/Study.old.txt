import { useCallback, useEffect, useRef, useState } from "react";
import { Link, useSearchParams } from "react-router-dom";
import YouTube, { type YouTubeProps } from "react-youtube";
import toast from "react-hot-toast";
import { supabase } from "../lib/supabaseClient";
import {
  ArrowLeft, BookOpen, Check, Clock3, FilePlus2, FileText, LoaderCircle,
  PlayCircle, Save, Search, Trash2, Pencil, X,
} from "lucide-react";

interface StudyNote {
  id: string;
  user_id: string;
  video_id: string | null;
  video_title: string | null;
  video_url: string | null;
  title: string;
  content: string;
  created_at: string;
  updated_at: string;
}
interface NoteTimestamp {
  id: string;
  note_id: string;
  seconds: number;
  label: string;
}
interface YouTubePlayer {
  seekTo: (seconds: number, allowSeekAhead?: boolean) => void;
  playVideo: () => void;
  getCurrentTime: () => number;
}
const inputClass = "w-full rounded-xl border border-zinc-700 bg-zinc-950 px-3 py-2.5 text-sm text-zinc-100 outline-none placeholder:text-zinc-600 focus:border-sky-500 focus:ring-2 focus:ring-sky-500/20";
const primaryClass = "inline-flex items-center justify-center gap-2 rounded-xl bg-sky-400 px-4 py-2.5 text-sm font-semibold text-zinc-950 transition hover:bg-sky-300 disabled:cursor-not-allowed disabled:opacity-50";
const secondaryClass = "inline-flex items-center justify-center gap-2 rounded-xl border border-zinc-700 bg-zinc-800 px-3 py-2.5 text-sm font-medium text-zinc-200 transition hover:bg-zinc-700 disabled:opacity-50";

export function Study() {
  const [searchParams] = useSearchParams();
  const videoId = (searchParams.get("videoId") ?? "").trim() || null;
  const videoTitle = searchParams.get("title") ?? "Untitled video";
  const [userId, setUserId] = useState("");
  const [notes, setNotes] = useState<StudyNote[]>([]);
  const [activeNoteId, setActiveNoteId] = useState("");
  const [noteTitle, setNoteTitle] = useState("");
  const [content, setContent] = useState("");
  const [search, setSearch] = useState("");
  const [timestamps, setTimestamps] = useState<NoteTimestamp[]>([]);
  const [timestampLabel, setTimestampLabel] = useState("");
  const [player, setPlayer] = useState<YouTubePlayer | null>(null);
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [addingTimestamp, setAddingTimestamp] = useState(false);
  const [deletingNote, setDeletingNote] = useState(false);
  const [playerError, setPlayerError] = useState("");
  const [lastSaved, setLastSaved] = useState("");
  const [creatingNote, setCreatingNote] = useState(false);
  const saveTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
  const dirtyRef = useRef(false);

  const formatTime = (n: number) => {
    const s = Math.max(0, Math.floor(n));
    const h = Math.floor(s / 3600), m = Math.floor((s % 3600) / 60), r = s % 60;
    return h ? `${String(h).padStart(2,"0")}:${String(m).padStart(2,"0")}:${String(r).padStart(2,"0")}` : `${String(m).padStart(2,"0")}:${String(r).padStart(2,"0")}`;
  };
  const seekTo = useCallback((seconds: number) => {
    if (!player) { toast.error("Wait for the video player to load."); return; }
    player.seekTo(seconds, true); player.playVideo();
  }, [player]);

  const fetchTimestamps = useCallback(async (noteId: string, uid: string) => {
    const { data, error } = await supabase.from("note_timestamps").select("id, note_id, seconds, label").eq("note_id", noteId).eq("user_id", uid).order("seconds", { ascending: true });
    if (error) throw error;
    setTimestamps((data ?? []) as NoteTimestamp[]);
  }, []);

  const selectNote = useCallback(async (note: StudyNote) => {
    setActiveNoteId(note.id);
    setNoteTitle(note.title || "Untitled note");
    setContent(note.content || "");
    dirtyRef.current = false;
    setLastSaved(note.updated_at ? new Date(note.updated_at).toLocaleString() : "");
    try { await fetchTimestamps(note.id, note.user_id); }
    catch (e) { console.error(e); toast.error("Could not load timestamps."); setTimestamps([]); }
  }, [fetchTimestamps]);

  const loadWorkspace = useCallback(async () => {
    setLoading(true);
    try {
      const { data: { user }, error: authError } = await supabase.auth.getUser();
      if (authError) throw authError;
      if (!user) { setUserId(""); setNotes([]); setActiveNoteId(""); toast.error("Please sign in to use Study Notes."); return; }
      setUserId(user.id);
      const { data, error } = await supabase.from("video_notes")
        .select("id, user_id, video_id, video_title, video_url, title, content, created_at, updated_at")
        .eq("user_id", user.id).order("updated_at", { ascending: false });
      if (error) throw error;
      const allNotes = (data ?? []) as StudyNote[];
      setNotes(allNotes);
      const preferred = allNotes.find(n => videoId && n.video_id === videoId) ?? allNotes[0];
      if (preferred) await selectNote(preferred);
      else {
        setActiveNoteId(""); setNoteTitle(videoId ? `${videoTitle} notes` : "My first note");
        setContent(""); setTimestamps([]); dirtyRef.current = false; setLastSaved("");
      }
    } catch (e) {
      console.error("Failed to load notes:", e);
      toast.error(e instanceof Error ? `Could not load notes: ${e.message}` : "Could not load notes.");
    } finally { setLoading(false); }
  }, [videoId, videoTitle, selectNote]);

  useEffect(() => { void loadWorkspace(); }, [loadWorkspace]);
  useEffect(() => () => { if (saveTimer.current) clearTimeout(saveTimer.current); }, []);

  const saveActiveNote = useCallback(async (showToast = true) => {
    if (!userId) { if (showToast) toast.error("Please sign in to save notes."); return false; }
    if (!noteTitle.trim()) { if (showToast) toast.error("Give this note a title first."); return false; }
    setSaving(true);
    try {
      const payload = {
        user_id: userId,
        video_id: videoId,
        video_title: videoId ? videoTitle : null,
        video_url: videoId ? `https://www.youtube.com/watch?v=${videoId}` : null,
        title: noteTitle.trim(),
        content,
        updated_at: new Date().toISOString(),
      };
      let result;
      if (activeNoteId) {
        result = await supabase.from("video_notes").update(payload).eq("id", activeNoteId).eq("user_id", userId)
          .select("id, user_id, video_id, video_title, video_url, title, content, created_at, updated_at").single();
      } else {
        result = await supabase.from("video_notes").insert(payload)
          .select("id, user_id, video_id, video_title, video_url, title, content, created_at, updated_at").single();
      }
      if (result.error) throw result.error;
      const saved = result.data as StudyNote;
      setActiveNoteId(saved.id);
      setNotes(prev => [saved, ...prev.filter(n => n.id !== saved.id)].sort((a,b) => new Date(b.updated_at).getTime() - new Date(a.updated_at).getTime()));
      dirtyRef.current = false;
      setLastSaved(new Date(saved.updated_at).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" }));
      if (showToast) toast.success("Note saved.");
      return true;
    } catch (e) {
      console.error("Failed to save note:", e);
      if (showToast) toast.error(e instanceof Error ? `Could not save note: ${e.message}` : "Could not save note.");
      return false;
    } finally { setSaving(false); }
  }, [userId, noteTitle, content, activeNoteId, videoId, videoTitle]);

  const updateTitle = (value: string) => { setNoteTitle(value); dirtyRef.current = true; };
  const updateContent = (value: string) => { setContent(value); dirtyRef.current = true; };
  useEffect(() => {
    if (!dirtyRef.current || loading || !userId) return;
    if (saveTimer.current) clearTimeout(saveTimer.current);
    saveTimer.current = setTimeout(() => { void saveActiveNote(false); }, 1200);
    return () => { if (saveTimer.current) clearTimeout(saveTimer.current); };
  }, [noteTitle, content, userId, loading, saveActiveNote]);

  const createNote = async () => {
    if (dirtyRef.current && activeNoteId) {
      const ok = await saveActiveNote(false);
      if (!ok) return;
    }
    setCreatingNote(true);
    const title = videoId ? `${videoTitle} notes` : "Untitled note";
    try {
      const { data, error } = await supabase.from("video_notes").insert({
        user_id: userId, video_id: videoId, video_title: videoId ? videoTitle : null,
        video_url: videoId ? `https://www.youtube.com/watch?v=${videoId}` : null,
        title, content: "", updated_at: new Date().toISOString(),
      }).select("id, user_id, video_id, video_title, video_url, title, content, created_at, updated_at").single();
      if (error) throw error;
      const created = data as StudyNote;
      setNotes(prev => [created, ...prev]);
      setActiveNoteId(created.id); setNoteTitle(created.title); setContent(""); setTimestamps([]);
      dirtyRef.current = false; setLastSaved("Just created");
      toast.success("New note created.");
    } catch (e) {
      console.error(e);
      toast.error(e instanceof Error ? `Could not create note: ${e.message}` : "Could not create note.");
    } finally { setCreatingNote(false); }
  };

  const deleteNote = async () => {
    if (!activeNoteId || !userId) return;
    if (!window.confirm("Delete this note and its saved timestamps? This cannot be undone.")) return;
    setDeletingNote(true);
    try {
      const { error } = await supabase.from("video_notes").delete().eq("id", activeNoteId).eq("user_id", userId);
      if (error) throw error;
      const remaining = notes.filter(n => n.id !== activeNoteId);
      setNotes(remaining); setActiveNoteId(""); setTimestamps([]); setContent(""); setLastSaved(""); dirtyRef.current = false;
      if (remaining.length) await selectNote(remaining[0]);
      else setNoteTitle(videoId ? `${videoTitle} notes` : "My first note");
      toast.success("Note deleted.");
    } catch (e) {
      console.error(e); toast.error(e instanceof Error ? `Could not delete note: ${e.message}` : "Could not delete note.");
    } finally { setDeletingNote(false); }
  };

  const addTimestamp = async () => {
    if (!player) { toast.error("Wait for the video player to load."); return; }
    if (!userId) { toast.error("Please sign in first."); return; }
    let currentNoteId = activeNoteId;
    if (!currentNoteId) {
      if (!noteTitle.trim()) { toast.error("Give this note a title first."); return; }
      const { data: created, error: createError } = await supabase.from("video_notes").insert({
        user_id: userId, video_id: videoId, video_title: videoTitle,
        video_url: `https://www.youtube.com/watch?v=${videoId}`,
        title: noteTitle.trim(), content, updated_at: new Date().toISOString(),
      }).select("id, user_id, video_id, video_title, video_url, title, content, created_at, updated_at").single();
      if (createError) { toast.error(`Could not create note: ${createError.message}`); return; }
      const saved = created as StudyNote;
      currentNoteId = saved.id;
      setActiveNoteId(saved.id);
      setNotes(prev => [saved, ...prev.filter(n => n.id !== saved.id)]);
      dirtyRef.current = false;
      setLastSaved("Just saved");
    }
    setAddingTimestamp(true);
    try {
      const seconds = Math.floor(player.getCurrentTime());
      const { data, error } = await supabase.from("note_timestamps").insert({
        note_id: currentNoteId, user_id: userId, seconds, label: timestampLabel.trim(),
      }).select("id, note_id, seconds, label").single();
      if (error) throw error;
      setTimestamps(prev => [...prev, data as NoteTimestamp].sort((a,b) => a.seconds - b.seconds));
      setTimestampLabel(""); toast.success(`Timestamp ${formatTime(seconds)} added.`);
    } catch (e) {
      console.error(e); toast.error(e instanceof Error ? `Could not add timestamp: ${e.message}` : "Could not add timestamp.");
    } finally { setAddingTimestamp(false); }
  };

  const deleteTimestamp = async (id: string) => {
    try {
      const { error } = await supabase.from("note_timestamps").delete().eq("id", id).eq("user_id", userId);
      if (error) throw error;
      setTimestamps(prev => prev.filter(t => t.id !== id)); toast.success("Timestamp deleted.");
    } catch (e) { console.error(e); toast.error("Could not delete timestamp."); }
  };

  const renderPreview = () => content.split(/(\b\d{1,2}:\d{2}(?::\d{2})?\b)/g).map((part, i) => {
    if (/^\d{1,2}:\d{2}(?::\d{2})?$/.test(part)) {
      const v = part.split(":").map(Number);
      const sec = v.length === 3 ? v[0]*3600 + v[1]*60 + v[2] : v[0]*60 + v[1];
      if (v[v.length - 1] < 60 && (v.length === 2 || v[1] < 60)) return <button key={`${part}-${i}`} type="button" onClick={() => seekTo(sec)} className="rounded bg-sky-500/10 px-1 font-semibold text-sky-300 underline">{part}</button>;
    }
    return <span key={i}>{part}</span>;
  });
  const playerOptions: YouTubeProps["opts"] = { width: "100%", height: "100%", playerVars: { autoplay: 0, rel: 0 } };
  const visibleNotes = notes.filter(n => `${n.title} ${n.content} ${n.video_title ?? ""}`.toLowerCase().includes(search.toLowerCase()));

  return (
    <main className="min-h-screen bg-[#09090b] px-3 py-5 text-zinc-100 sm:px-5 lg:px-8 lg:py-7">
      <div className="mx-auto max-w-[1600px]">
        <header className="mb-5 flex flex-wrap items-center justify-between gap-3">
          <div className="flex items-center gap-3"><div className="flex h-11 w-11 items-center justify-center rounded-2xl bg-sky-400/10 text-sky-300"><BookOpen size={22}/></div><div><p className="text-xs font-semibold uppercase tracking-[.2em] text-sky-300">FocusTube</p><h1 className="text-xl font-bold sm:text-2xl">Your notes workspace</h1></div></div>
          <Link to="/Feed" className={secondaryClass}><ArrowLeft size={16}/> Back to Feed</Link>
        </header>

        <div className="grid grid-cols-1 gap-5 lg:grid-cols-[260px_minmax(0,1fr)]">
          <aside className="min-w-0 rounded-2xl border border-zinc-800 bg-zinc-900 p-3">
            <button onClick={() => void createNote()} disabled={!userId || loading || creatingNote} className={`${primaryClass} w-full`}><FilePlus2 size={16}/>{creatingNote ? "Creating…" : "New note"}</button>
            <div className="relative my-3"><Search size={15} className="absolute left-3 top-3 text-zinc-500"/><input value={search} onChange={e => setSearch(e.target.value)} placeholder="Search your notes" className={`${inputClass} pl-9`}/></div>
            <div className="mb-2 flex items-center justify-between px-1 text-xs text-zinc-500"><span>ALL NOTES</span><span>{visibleNotes.length}</span></div>
            <div className="max-h-[50vh] space-y-1 overflow-y-auto lg:max-h-[calc(100vh-280px)]">
              {loading ? <p className="p-3 text-sm text-zinc-500">Loading notes…</p> : visibleNotes.length ? visibleNotes.map(n => (
                <button key={n.id} onClick={async () => { if (dirtyRef.current) { const ok = await saveActiveNote(false); if (!ok) return; } await selectNote(n); }} className={`w-full rounded-xl p-3 text-left transition ${activeNoteId === n.id ? "border border-sky-500/30 bg-sky-500/10" : "border border-transparent hover:bg-zinc-800"}`}>
                  <span className="flex items-start gap-2"><FileText size={16} className={`mt-0.5 shrink-0 ${activeNoteId === n.id ? "text-sky-300" : "text-zinc-500"}`}/><span className="min-w-0 flex-1"><span className="block truncate text-sm font-medium text-zinc-200">{n.title || "Untitled note"}</span><span className="mt-1 block truncate text-xs text-zinc-500">{n.video_title || "Personal note"}</span><span className="mt-1 block text-[11px] text-zinc-600">{n.updated_at ? new Date(n.updated_at).toLocaleDateString() : ""}</span></span></span>
                </button>
              )) : <p className="px-3 py-6 text-center text-sm text-zinc-500">No notes yet. Create your first note.</p>}
            </div>
          </aside>

          <section className="min-w-0">
            {loading ? <div className="flex min-h-72 items-center justify-center gap-2 text-zinc-500"><LoaderCircle className="animate-spin"/> Loading workspace…</div> : (
              <div className="grid grid-cols-1 items-start gap-5 xl:grid-cols-[minmax(0,1.05fr)_minmax(360px,.95fr)]">
                <div className="min-w-0 space-y-5">
                  <section className="overflow-hidden rounded-2xl border border-zinc-800 bg-zinc-900">
                    <div className="flex items-center gap-2 border-b border-zinc-800 px-4 py-3"><PlayCircle className="text-sky-300" size={18}/><h2 className="font-semibold">Video lesson</h2></div>
                    {videoId ? <div className="aspect-video bg-black"><YouTube videoId={videoId} opts={playerOptions} onReady={e => { setPlayer(e.target as YouTubePlayer); setPlayerError(""); }} onError={e => { console.error("YouTube player error:", e.data); setPlayerError("This video may not allow embedded playback."); }} className="h-full w-full" iframeClassName="h-full w-full"/></div> : <div className="flex aspect-video flex-col items-center justify-center gap-2 bg-zinc-950 p-6 text-center"><PlayCircle size={32} className="text-zinc-600"/><p className="text-sm text-zinc-400">No video selected</p><Link to="/Feed" className="text-sm text-sky-300 underline">Choose a video from your Feed</Link></div>}
                    {playerError && <div className="p-3 text-sm text-amber-200">{playerError} <a href={`https://www.youtube.com/watch?v=${encodeURIComponent(videoId ?? "")}`} target="_blank" rel="noreferrer" className="underline">Open on YouTube</a></div>}
                  </section>

                  {videoId && <section className="rounded-2xl border border-zinc-800 bg-zinc-900 p-4">
                    <div className="mb-3 flex items-center justify-between"><div className="flex items-center gap-2"><Clock3 size={18} className="text-violet-300"/><h2 className="font-semibold">Key moments</h2></div><span className="text-xs text-zinc-500">{timestamps.length} saved</span></div>
                    <div className="flex flex-col gap-2 sm:flex-row"><input value={timestampLabel} onChange={e => setTimestampLabel(e.target.value)} onKeyDown={e => { if(e.key === "Enter"){e.preventDefault(); void addTimestamp();}}} placeholder="Label this moment (optional)" maxLength={200} className={`${inputClass} min-w-0 flex-1`}/><button onClick={() => void addTimestamp()} disabled={!player || addingTimestamp} className={primaryClass}>{addingTimestamp ? <LoaderCircle size={16} className="animate-spin"/> : <Clock3 size={16}/>} Add current time</button></div>
                    <ul className="mt-4 space-y-2">{timestamps.map(t => <li key={t.id} className="flex items-center gap-3 rounded-xl border border-zinc-800 bg-zinc-950 p-3"><button onClick={() => seekTo(t.seconds)} className="rounded-lg bg-sky-500/10 px-3 py-2 font-mono text-sm font-bold text-sky-300">{formatTime(t.seconds)}</button><span className="min-w-0 flex-1 break-words text-sm text-zinc-300">{t.label || "Important moment"}</span><button onClick={() => void deleteTimestamp(t.id)} className="rounded-lg p-2 text-zinc-500 hover:text-red-300" aria-label="Delete timestamp"><Trash2 size={16}/></button></li>)}</ul>
                    {!timestamps.length && <p className="mt-4 rounded-xl border border-dashed border-zinc-700 p-4 text-center text-sm text-zinc-500">Add a timestamp while watching to save an important moment.</p>}
                  </section>}
                </div>

                <div className="min-w-0 space-y-5">
                  {activeNoteId || notes.length === 0 ? (
                    <section className="overflow-hidden rounded-2xl border border-zinc-800 bg-zinc-900">
                      <div className="flex flex-wrap items-center gap-2 border-b border-zinc-800 px-4 py-3"><FileText size={18} className="text-emerald-300"/><span className="font-semibold">Note editor</span><span className="ml-auto flex items-center gap-1 text-xs text-zinc-500">{saving ? <LoaderCircle size={13} className="animate-spin"/> : <Check size={13} className="text-emerald-400"/>}{saving ? "Saving…" : dirtyRef.current ? "Unsaved changes" : lastSaved ? `Saved ${lastSaved}` : "Not saved yet"}</span></div>
                      <div className="space-y-4 p-4">
                        <input value={noteTitle} onChange={e => updateTitle(e.target.value)} maxLength={150} className={`${inputClass} text-base font-semibold`} placeholder="Give your note a title"/>
                        <textarea value={content} onChange={e => updateContent(e.target.value)} rows={16} className={`${inputClass} min-h-[320px] resize-y leading-7`} placeholder={"Write your notes here…\n\nYou can add timestamps like 02:35 and click them in the preview."}/>
                        <div className="flex flex-wrap items-center justify-between gap-2"><span className="text-xs text-zinc-600">{content.length} characters · Autosaves after you pause</span><div className="flex gap-2"><button onClick={() => void deleteNote()} disabled={!activeNoteId || deletingNote} className={secondaryClass}><Trash2 size={15}/>{deletingNote ? "Deleting…" : "Delete"}</button><button onClick={() => void saveActiveNote(true)} disabled={saving || !userId} className={primaryClass}><Save size={15}/>Save note</button></div></div>
                      </div>
                    </section>
                  ) : (
                    <div className="rounded-2xl border border-dashed border-zinc-700 bg-zinc-900/50 p-8 text-center"><FileText size={28} className="mx-auto mb-3 text-zinc-600"/><p className="font-medium">Choose a note to edit</p><p className="mt-1 text-sm text-zinc-500">Or create a new one from the sidebar.</p></div>
                  )}
                  <section className="rounded-2xl border border-zinc-800 bg-zinc-900 p-4"><div className="mb-3 flex items-center gap-2"><BookOpen size={17} className="text-sky-300"/><h2 className="font-semibold">Live preview</h2></div><div className="min-h-24 whitespace-pre-wrap break-words rounded-xl border border-zinc-800 bg-zinc-950 p-4 text-sm leading-7 text-zinc-300">{content.trim() ? renderPreview() : <span className="text-zinc-600">Your note preview will appear here as you type.</span>}</div><p className="mt-2 text-xs text-zinc-600">Timestamps such as 02:35 are clickable when a video is selected.</p></section>
                </div>
              </div>
            )}
          </section>
        </div>
        <footer className="mt-6 border-t border-zinc-800 pt-4 text-xs text-zinc-600">FocusTube · Your personal learning library</footer>
      </div>
    </main>
  );
}
export default Study;
